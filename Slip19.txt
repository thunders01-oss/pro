Q1. Create a dataframe containing information about different vehicles such as Fuel_Efficiency, 
Engine_Power, Vehicle_Weight, Engine_Size, and Acceleration. Add 5 records in dataframe, 
including outlier. Create Box Plots for Fuel Efficiency and Engine Power and identify outliers.



import pandas as pd
import matplotlib.pyplot as plt

# Create DataFrame
data = {
    'Fuel_Efficiency': [15, 18, 20, 22, 80],
    'Engine_Power': [100, 120, 140, 160, 500],
    'Vehicle_Weight': [1000, 1200, 1300, 1400, 1500],
    'Engine_Size': [1000, 1200, 1400, 1600, 1800],
    'Acceleration': [12, 11, 10, 9, 8]
}

df = pd.DataFrame(data)

print("Vehicle Data:")
print(df)

# Create Box Plots
plt.figure(figsize=(10, 5))

plt.subplot(1, 2, 1)
plt.boxplot(df['Fuel_Efficiency'])
plt.title("Fuel Efficiency")
plt.ylabel("Fuel Efficiency")

plt.subplot(1, 2, 2)
plt.boxplot(df['Engine_Power'])
plt.title("Engine Power")
plt.ylabel("Engine Power")

plt.tight_layout()
plt.show()

# Detect Outliers using IQR
for col in ['Fuel_Efficiency', 'Engine_Power']:
    Q1 = df[col].quantile(0.25)
    Q3 = df[col].quantile(0.75)
    IQR = Q3 - Q1

    lower = Q1 - 1.5 * IQR
    upper = Q3 + 1.5 * IQR

    outliers = df[(df[col] < lower) | (df[col] > upper)]

    print("\nOutliers in", col, ":")
    print(outliers[[col]])






Q2. Write a Python program to create a DataFrame with the attributes Temperature, Humidity, Wind 
Speed, Pressure, and Rainfall. Apply the following data transformation techniques and display 
the transformed data:
a)Min-MaxScaling 
b)Standardization
c) Normalization



import pandas as pd
from sklearn.preprocessing import MinMaxScaler
from sklearn.preprocessing import StandardScaler
from sklearn.preprocessing import Normalizer

# Create DataFrame
data = {
    'Temperature': [25, 30, 35, 40, 45],
    'Humidity': [40, 50, 60, 70, 80],
    'Wind Speed': [5, 10, 15, 20, 25],
    'Pressure': [1000, 1010, 1020, 1030, 1040],
    'Rainfall': [10, 20, 30, 40, 50]
}

df = pd.DataFrame(data)

print("Original Data:")
print(df)

# a) Min-Max Scaling
mms = MinMaxScaler()
a = mms.fit_transform(df)

print("\nMin-Max Scaling:")
print(pd.DataFrame(a, columns=df.columns).round(2))

# b) Standardization
ss = StandardScaler()
b = ss.fit_transform(df)

print("\nStandardization:")
print(pd.DataFrame(b, columns=df.columns).round(2))

# c) Normalization
norm = Normalizer()
c = norm.fit_transform(df)

print("\nNormalization:")
print(pd.DataFrame(c, columns=df.columns).round(4))
