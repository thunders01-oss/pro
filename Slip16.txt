Q1. Create a DataFrame containing student performance data with attributes such as Study Hours, 
Attendance, Assignment Score, and Exam Score. Calculate the correlation matrix for these 
attributes and generate a Heat Map to visualize the correlation between them. Display the 
correlation values, suitable labels, and an appropriate title.


import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

# Create DataFrame
data = {
    'Study Hours': [2, 3, 4, 5, 6, 7],
    'Attendance': [60, 65, 70, 75, 80, 90],
    'Assignment Score': [50, 55, 65, 70, 80, 90],
    'Exam Score': [45, 50, 60, 65, 80, 95]
}

df = pd.DataFrame(data)

print("Student Performance Data:")
print(df)

# Calculate correlation matrix
corr = df.corr()

print("\nCorrelation Matrix:")
print(corr)

# Generate Heat Map
sns.heatmap(corr, annot=True, cmap='coolwarm',
            fmt='.2f', linewidths=0.5)

plt.title("Correlation Heat Map of Student Performance")
plt.show()




Q2. Write a Python program to create a DataFrame with the attributes Temperature, Humidity, Wind-
Speed, Pressure, and Rainfall. Apply the following data transformation techniques and display
the transformed data:
a)Min-MaxScaling 
b)Standardization
c) Normalization


import pandas as pd
from sklearn.preprocessing import MinMaxScaler
from sklearn.preprocessing import StandardScaler
from sklearn.preprocessing import Normalizer

# Create DataFrame
data = {
    'Temperature': [25, 30, 35, 40, 45],
    'Humidity': [40, 50, 60, 70, 80],
    'Wind-Speed': [5, 10, 15, 20, 25],
    'Pressure': [1000, 1010, 1020, 1030, 1040],
    'Rainfall': [10, 20, 30, 40, 50]
}

df = pd.DataFrame(data)

print("Original Data:")
print(df)

# a) Min-Max Scaling
minmax = MinMaxScaler()
scaled_data = minmax.fit_transform(df)

print("\nMin-Max Scaling:")
print(pd.DataFrame(scaled_data, columns=df.columns).round(2))

# b) Standardization
standard = StandardScaler()
standard_data = standard.fit_transform(df)

print("\nStandardization:")
print(pd.DataFrame(standard_data, columns=df.columns).round(2))

# c) Normalization
normalizer = Normalizer()
normalized_data = normalizer.fit_transform(df)

print("\nNormalization:")
print(pd.DataFrame(normalized_data, columns=df.columns).round(2))
